11def find_mins_maxs( 

12 points: np.ndarray, enlargement: Optional[float] = 0 

13) -> Tuple[np.ndarray, np.ndarray]: 

14 """Computes the axis-aligned bounding box for a set of 3D points. 

15 

16 This function calculates the minimum and maximum coordinate values along 

17 each axis (X, Y, Z) to determine the bounding box of the input points. 

18 An optional enlargement factor can be applied to expand the bounding box 

19 in all directions. 

20 

21 Args: 

22 points (np.ndarray): A NumPy array of shape (N, 3) representing N 

23 points in 3D space. 

24 enlargement (Optional[float]): An optional value to expand the bounding 

25 box equally along all axes. Defaults to 0. 

26 

27 Returns: 

28 Tuple[np.ndarray, np.ndarray]: Two NumPy arrays representing the 

29 minimum and maximum coordinates of the bounding box. The first 

30 array contains the minimum values [min_x, min_y, min_z], and the 

31 second array contains the maximum values [max_x, max_y, max_z]. 

32 """ 

33 min_coords = np.min(points, axis=0) - enlargement 

34 max_coords = np.max(points, axis=0) + enlargement 

35 

36 return min_coords, max_coords

39def normalize_values(data: np.ndarray, pixel_range: np.ndarray) -> np.ndarray: 

40 """Normalizes data to a range between 0 and 1 based on the provided pixel 

41 range. 

42 

43 This function shifts the data by subtracting the minimum pixel range value 

44 and then scales it by dividing it by the total range. The resulting values 

45 will be in the range [0, 1]. 

46 

47 Arguments: 

48 data (np.ndarray): The array of data values to normalize. 

49 pixel_range (np.ndarray): A NumPy array containing the minimum and 

50 maximum pixel range values [min, max] used for normalization. 

51 

52 Returns: 

53 np.ndarray: The normalized data with values scaled between 0 and 1. 

54 """ 

55 # Validate supplied pixel range 

56 range_diff = pixel_range[1] - pixel_range[0] 

57 if range_diff == 0: 

58 logging.error("Pixel range difference is zero.") 

59 elif range_diff < 0: 

60 logging.error("Pixel range is inverted (max < min).") 

61 

62 # Normalize the data 

63 normalized_data = (data - pixel_range[0]) / range_diff 

64 

65 return normalized_data

91def smooth_data( 

92 data: np.ndarray, 

93 smoothing_window: int, 

94 mask: np.ndarray, 

95) -> np.ndarray: 

96 """Applies a smoothing filter to 3D image data by averaging pixel values. 

97 

98 This function reduces noise or measurement errors in the image data by 

99 applying a smoothing filter. The filter calculates the average pixel value 

100 within a neighborhood defined by the `smoothing_window` parameter, which 

101 helps to smooth out irregularities in the data. 

102 

103 If a mask is provided, performs masked smoothing by: 

104 - Zero data outside mask → data_masked 

105 - Smooth the zeroed data → smoothed_data (artificially darkened at edges) 

106 - Smooth the mask → smoothed_mask (tells you the "weight" of valid data) 

107 - Divide smoothed data by smoothed mask 

108 → This "undoes" the darkening by renormalizing 

109 - Only update pixels that were originally inside the mask 

110 → Preserves original data outside 

111 

112 Args: 

113 data (np.ndarray): A 3D array containing the pixel data to be smoothed. 

114 smoothing_window (int): The size of the neighborhood (in points) used 

115 to compute the average. Larger values result in smoother data, 

116 but may reduce fine details. 

117 mask (np.ndarray): A boolean mask array of the same shape as `data`. 

118 If provided, smoothing is only applied within the 

119 masked region. 

120 

121 Returns: 

122 np.ndarray: The smoothed image data as a 3D array, where each pixel's 

123 value has been replaced by the average of its neighbors within the 

124 defined smoothing window. 

125 """ 

126 logging.info("Smooth data!") 

127 

128 if mask is None or not np.any(mask): 

129 logging.warning( 

130 "No mask provided for smoothing. " 

131 "Smoothing will be applied to entire data." 

132 ) 

133 return uniform_filter( 

134 data, size=smoothing_window, mode="nearest", axes=(0, 1, 2) 

135 ) 

136 

137 orig_dtype = data.dtype 

138 data_f = data.astype(np.float32, copy=False) 

139 

140 # Create a copy of the data where all values outside the mask are zero. 

141 data_masked = data_f.copy() 

142 data_masked[~mask, ...] = 0 

143 

144 # Apply a uniform filter to data_masked, which darkens values near edges. 

145 smoothed_data = uniform_filter( 

146 data_masked, size=smoothing_window, mode="nearest", axes=(0, 1, 2) 

147 ) 

148 # Apply the same filter to the mask to find the proportion of valid data. 

149 smoothed_mask = uniform_filter( 

150 mask.astype(np.float32), 

151 size=smoothing_window, 

152 mode="nearest", 

153 axes=(0, 1, 2), 

154 ) 

155 

156 # Reshape smoothed_mask to match smoothed_data for broadcasting 

157 target_shape = list(smoothed_mask.shape) + [1] * ( 

158 smoothed_data.ndim - smoothed_mask.ndim 

159 ) 

160 smoothed_mask_exp = smoothed_mask.reshape(target_shape) 

161 

162 valid_mask = (mask) & (smoothed_mask > 1e-8) 

163 result_f = data_f.copy() 

164 

165 normalized = np.zeros_like(smoothed_data, dtype=np.float32) 

166 # Normalize smoothed data to correct darkening at the edges. 

167 np.divide( 

168 smoothed_data, 

169 smoothed_mask_exp, 

170 out=normalized, 

171 where=smoothed_mask_exp > 1e-8, 

172 ) 

173 # Update pixels inside the original mask with the smoothed values. 

174 result_f[valid_mask, ...] = normalized[valid_mask, ...] 

175 

176 return result_f.astype(orig_dtype, copy=False)

179def make_json_serializable(obj: Any) -> Any: 

180 """Converts NumPy data types to standard Python types for JSON 

181 serialization. This function recursively processes NumPy arrays and scalar 

182 types to ensure compatibility with JSON serialization. 

183 

184 Args: 

185 obj: The object to convert, which can be a NumPy array, scalar, or 

186 other data type. 

187 Returns: 

188 Any: The converted object, where NumPy arrays are converted to lists, 

189 and NumPy scalars are converted to standard Python types (int, float). 

190 """ 

191 if isinstance(obj, np.ndarray): 

192 return [make_json_serializable(item) for item in obj] 

193 elif isinstance( 

194 obj, 

195 ( 

196 np.int64, 

197 np.int32, 

198 np.int16, 

199 np.int8, 

200 np.uint64, 

201 np.uint32, 

202 np.uint16, 

203 np.uint8, 

204 ), 

205 ): 

206 return int(obj) 

207 elif isinstance(obj, (np.float64, np.float32, np.float16)): 

208 return float(obj) 

209 return obj

212def world_to_grid_coords( 

213 points_world: np.ndarray, orientation: np.ndarray, position: np.ndarray 

214) -> np.ndarray: 

215 """Convert world coordinates to image grid coordinates using image 

216 orientation and origin. 

217 

218 Args: 

219 points_world (np.ndarray): A NumPy array of shape (N, 3) representing 

220 N points in world coordinates. 

221 orientation (np.ndarray): A 3x3 matrix representing the orientation 

222 of the image grid in world space. 

223 position (np.ndarray): A 1D array of shape (3,) representing the origin 

224 of the image grid in world space. 

225 

226 Returns: 

227 np.ndarray: A NumPy array of shape (N, 3) representing the points in 

228 image grid coordinates. 

229 """ 

230 return np.linalg.solve(orientation, (points_world - position).T).T

233def _grid_bbox_indices_for_points( 

234 grid_points: np.ndarray, grid_coords 

235) -> Tuple[int, int, int, int, int, int]: 

236 """Compute axis-aligned grid bounding box indices for given grid points. 

237 

238 Args: 

239 grid_points (np.ndarray): A NumPy array of shape (N, 3) 

240 representing N points in grid coordinates. 

241 grid_coords: An object containing the grid's coordinate 

242 arrays (slice, row, col). 

243 

244 Returns: 

245 Tuple[int, int, int, int, int, int]: A tuple of six integers representing 

246 the bounding box indices (z0, z1, y0, y1, x0, x1), where 

247 - z0, z1: Start and end indices along the Z-axis. 

248 - y0, y1: Start and end indices along the Y-axis. 

249 - x0, x1: Start and end indices along the X-axis. 

250 """ 

251 mins = np.min(grid_points, axis=0) 

252 maxs = np.max(grid_points, axis=0) 

253 

254 # Use np.clip to avoid redundant max/min calls 

255 z0 = np.clip( 

256 np.searchsorted(grid_coords.slice, mins[0], side="left"), 

257 0, 

258 len(grid_coords.slice), 

259 ) 

260 z1 = np.clip( 

261 np.searchsorted(grid_coords.slice, maxs[0], side="right"), 

262 0, 

263 len(grid_coords.slice), 

264 ) 

265 y0 = np.clip( 

266 np.searchsorted(grid_coords.row, mins[1], side="left"), 

267 0, 

268 len(grid_coords.row), 

269 ) 

270 y1 = np.clip( 

271 np.searchsorted(grid_coords.row, maxs[1], side="right"), 

272 0, 

273 len(grid_coords.row), 

274 ) 

275 x0 = np.clip( 

276 np.searchsorted(grid_coords.col, mins[2], side="left"), 

277 0, 

278 len(grid_coords.col), 

279 ) 

280 x1 = np.clip( 

281 np.searchsorted(grid_coords.col, maxs[2], side="right"), 

282 0, 

283 len(grid_coords.col), 

284 ) 

285 

286 return z0, z1, y0, y1, x0, x1

289def create_mesh_mask(discretization, image) -> np.ndarray: 

290 """Create a voxel-based mask of the discretization in the image's grid 

291 space. 

292 

293 This function generates a 3D boolean mask that represents a discretized 

294 mesh within the grid space of a given image. 

295 The mask is computed by determining which voxels in the image grid are 

296 enclosed by the convex hulls of the mesh elements. 

297 

298 Args: 

299 discretization: A discretized mesh object containing nodes, elements 

300 and surfaces. 

301 image: An image object that provides the grid space and pixel data. 

302 

303 Returns: 

304 np.ndarray: A 3D boolean array where `True` indicates that the 

305 corresponding voxel is part of the mesh mask, and `False` otherwise. 

306 

307 Notes: 

308 - The function uses convex hulls to approximate the spatial extent of 

309 each mesh element in the image grid space. 

310 - The convex hull computation requires at least three points. Elements 

311 with fewer than three nodes are skipped. 

312 - The mask is constructed iteratively by updating the relevant voxels 

313 for each mesh element. 

314 """ 

315 logging.info("Create mesh mask in image grid space") 

316 

317 mask = np.zeros(image.pixel_data.shape[:3], dtype=bool) 

318 

319 node_id_to_idx = {nid: i for i, nid in enumerate(discretization.nodes.ids)} 

320 nodes_grid = world_to_grid_coords( 

321 discretization.nodes.coords, image.orientation, image.position 

322 ) 

323 

324 for ele in discretization.elements: 

325 node_idxs = np.fromiter( 

326 (node_id_to_idx[nid] for nid in ele.node_ids), 

327 dtype=int, 

328 count=len(ele.node_ids), 

329 ) 

330 ele_node_grid = nodes_grid[node_idxs] 

331 

332 if ele_node_grid.shape[0] < 3: 

333 continue 

334 

335 try: 

336 hull = ConvexHull(ele_node_grid) 

337 except Exception: 

338 continue 

339 

340 z0, z1, y0, y1, x0, x1 = _grid_bbox_indices_for_points( 

341 ele_node_grid, image.grid_coords 

342 ) 

343 

344 zz, yy, xx = np.meshgrid( 

345 image.grid_coords.slice[z0:z1], 

346 image.grid_coords.row[y0:y1], 

347 image.grid_coords.col[x0:x1], 

348 indexing="ij", 

349 ) 

350 grid_points = np.stack((zz.ravel(), yy.ravel(), xx.ravel()), axis=-1) 

351 

352 A, b = hull.equations[:, :-1], hull.equations[:, -1] 

353 inside_mask = np.all(A @ grid_points.T + b[:, None] <= 0, axis=0) 

354 

355 inside_mask = inside_mask.reshape(zz.shape) 

356 mask[z0:z1, y0:y1, x0:x1] |= inside_mask 

357 

358 return mask
